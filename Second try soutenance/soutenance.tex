\begin{frame}{Qu'est-ce qu'un carnet d'ordres ?}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\centering
\includegraphics[width=\linewidth]{fig_lob_intro.png}

\vspace{0.3ex}
{\scriptsize Source : MODAL Queue Reactive (Lehalle, 2025-26).}
\end{column}
\begin{column}{0.44\textwidth}
\small
Un \textbf{carnet d'ordres à cours limité} centralise toutes les intentions d'achat et de vente en attente pour un actif donné.\\[0.8ex]

\textbf{Côté acheteur} (\textcolor{green!60!black}{bid}, $q_b$) : ordres d'achat à prix $\leq p_0$.\\[0.5ex]
\textbf{Côté vendeur} (\textcolor{red!70!black}{ask}, $q_a$) : ordres de vente à prix $\geq p_0$.\\[0.8ex]

\begin{itemize}\setlength\itemsep{0.3em}
  \item \textbf{Première limite} : la meilleure offre disponible de chaque côté ($Q_{-1}$ et $Q_1$).
  \item \textbf{Spread} : écart entre la première limite ask et la première limite bid.
  \item \textbf{Transaction} : un ordre de marché consomme la première limite du côté opposé — si elle s'épuise, le \alert{prix saute}.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

%================================================================
\begin{frame}{Le fil conducteur}
\begin{columns}[T]
\begin{column}{0.56\textwidth}
À chaque instant, le carnet contient un volume côté acheteur ($q_b$, le \emph{bid}) et côté vendeur ($q_a$, l'\emph{ask}). Dans le modèle Queue Reactive, le prix change lorsque la meilleure limite d'un des deux côtés est consommée.\\[1ex]

La question centrale se ramène donc à deux temps d'atteinte :
\[
  \taua=\inf\{t: q_{a,1}(t)=0\},\qquad \taub=\inf\{t: q_{b,1}(t)=0\}.
\]

\emph{Quel côté s'épuise en premier, à quel horizon, et avec quelle probabilité ?}
\end{column}
\begin{column}{0.44\textwidth}
\textbf{Plan de l'étude :}
\begin{itemize}
\item[\textbf{I.}] \textbf{Modéliser} la logique du Carnet d'Ordres — du Poisson au Hawkes couplé.
\item[\textbf{II.}] \textbf{Optimiser} les ressources computationnelles et \textbf{Simuler} quelques scénarios extrêmes.
\item[\textbf{III.}] \textbf{Calibrer} sur un vrai crash du marché.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[t]{Point de départ : modèle Poisson indépendant}
\small
\begin{columns}[T,onlytextwidth]

\begin{column}{0.54\textwidth}
Chaque file est d'abord modélisée par
\vspace{-0.2em}
\[
\begin{aligned}
q(t)&=q_0+N^+_t-N^-_t,\\
\mu&=\lambda^+-\lambda^-=-0{,}8<0,\\
\sigma^2&=\lambda^++\lambda^-=3.2.
\end{aligned}
\]

\vspace{-0.7em}
La limite diffusive donne un temps d'atteinte de 0 de loi
\alert{inverse-gaussienne} :
\vspace{-0.3em}
\[
\tau\sim \mathrm{IG}\!\left(\frac{q_0}{|\mu|},\frac{q_0^2}{\sigma^2}\right),
\qquad
\E[\tau]=12{,}5.
\]

\vspace{-0.5em}
\textbf{Rôle dans le projet :} étalon analytique pour valider la simulation.

\vspace{-0.3em}
\begin{itemize}
\setlength\itemsep{0.1em}
\item MC : $\E[\tau]=12{,}52$ ($+0{,}14\%$).
\item Test KS : $p=0{,}097$ ; IG non rejetée.
\end{itemize}
\end{column}

\begin{column}{0.44\textwidth}
\centering
\includegraphics[width=\linewidth]{fig_ig.png}

\vspace{-0.3em}
{\scriptsize Densité IG théorique vs histogramme simulé.}
\end{column}

\end{columns}
\end{frame}

\begin{frame}{Probabilité de ruine : référence par quadrature}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
Par conditionnement sur $\taua$ :
\[
\PP(\taua<\taub)=\int_0^\infty f_{\taua}(t)\,\bigl(1-F_{\taub}(t)\bigr)\,dt .
\]
La quadrature de Gauss--Legendre donne une référence déterministe pour comparer le Monte-Carlo.\\[1ex]
\textbf{Vérification numérique :} l'écart maximal avec Monte-Carlo est $0{,}049$, du même ordre que l'erreur statistique MC ($\approx 0{,}044$). La différence observée est donc compatible avec la variabilité de simulation.
\end{column}
\begin{column}{0.54\textwidth}
\centering
\includegraphics[width=\textwidth]{fig_ruine.png}\\
{\scriptsize MC (g.) vs quadrature IG (c.) vs écart (d.). Sur la diagonale $\hat p\approx\tfrac12$.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Processus de Hawkes : mémoire et troncature}
\begin{columns}[T,onlytextwidth]

\begin{column}{0.52\textwidth}
\small
L'intensité d'annulation intègre la mémoire des événements passés :
\vspace{-0.2em}
\[
\lambda^-(t)=\mu^-+\alpha\!\!\sum_{s\in\mathcal N^-,\,s<t}\!\! e^{-\beta(t-s)}
-\alpha\!\!\sum_{s\in\mathcal N^+,\,s<t}\!\! e^{-\beta(t-s)}.
\]

\vspace{-0.7em}
{\scriptsize Chaque événement imprime une impulsion sur $\lambda^-$, positive pour $\mathcal N^-$ et négative pour $\mathcal N^+$, puis cette contribution s'amortit exponentiellement.}\\[0.3ex]
Sans troncature, la théorie stationnaire donne $\bar\lambda^-_\infty=2{,}80$.\\[0.5ex]

\textbf{Simulation :} la moyenne mesurée est $2{,}96$, soit \alert{$+5{,}9\%$}.

\begin{block}{Lecture de l'écart}
\footnotesize
La théorie est calculée sans troncature, tandis que la simulation impose un plancher
$\lambda^-\geq 0$. Les inhibitions trop fortes sont donc coupées, ce qui explique la surestimation observée de $\bar\lambda^-$.
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\centering
\includegraphics[width=\linewidth]{fig_hawkes_bias.png}

\vspace{-0.3em}
{\scriptsize Moyennes temporelles de $\lambda^-$ ; ligne rouge $=2{,}80$. La masse est décalée à droite.}
\end{column}

\end{columns}
\end{frame}
%----------------------------------------------------------------
\begin{frame}{Hawkes couplé : effet du couplage bid/ask}
\begin{columns}[T]
\begin{column}{0.56\textwidth}
\small
Chaque côté conserve une mémoire décroissante : auto-excitation par les décréments, \alert{inhibition par les incréments}, et excitation croisée par l'autre file :
\[
\resizebox{\linewidth}{!}{$\displaystyle
\lma(t)=\mu_a^- +\alpha\!\!\sum_{s\in\mathcal D_a}\!e^{-\beta(t-s)}
\,\alert{-\,\alpha\!\!\sum_{s\in\mathcal A_a}\!e^{-\beta(t-s)}}
\,+\alpha\!\!\sum_{s\in\mathcal C_b}\!e^{-\beta(t-s)}$}
\]
{\scriptsize $\mathcal D_a,\mathcal A_a$ : décréments / incréments ask ; $\mathcal C_b=\mathcal D_b\cup\mathcal A_b$ : tous les événements bid.}\\[1ex]
\textbf{Effet mesuré} au point $q_0=(10,10)$ :
\[
\E[\min(\taua,\taub)]:\; 8{,}3\ \text{(Poisson)}\ \to\ 5{,}8\ \text{(couplé)},\quad -30\%.
\]
Le couplage accélère l'épuisement conjoint des deux côtés : il introduit un mécanisme de cascade absent du cas Poisson indépendant.
\end{column}
\begin{column}{0.44\textwidth}
\centering
\includegraphics[width=0.96\textwidth]{fig_couple3d.png}\\
{\scriptsize Hawkes couplé : temps moyen (g.) et probabilité ask avant bid (d.).}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Deuxième limite : reconstruire le carnet}
\begin{columns}[T]
\begin{column}{0.40\textwidth}
\footnotesize
\textbf{Deux lois pour la deuxième limite.} Le niveau 2 évolue avant le saut :
\[
\begin{aligned}
\lambda^+_{2,0}(t)&=\mu^+_2,\\
\lambda^+_{2,c}(t)&=\mu^+_2+
\sum_{T_i^\downarrow<t} a\,e^{-b(t-T_i^\downarrow)}.
\end{aligned}
\]
Sans couplage : référence autonome.\\
 Avec couplage : les décréments du niveau 1 rechargent $q_2$.

Quand l'ask s'épuise, le volume de niveau 2 devient la nouvelle meilleure limite :
\[
q_{a,1}\leftarrow q_{a,2}(\tau_a),\qquad
p_0\uparrow\tfrac{\delta^p}{2}.
\]

\textbf{Donc :} la \alert{loi conditionnelle} de $q_2(\tau)$ dépend du choix de couplage pour $\lambda_2^+$.
\end{column}
\begin{column}{0.60\textwidth}
\centering
\scriptsize\textbf{Sans couplage :} $\lambda_2^+=\mu_2^+$, moyennes $\approx 8{,}0$\\[-0.2ex]
\includegraphics[width=0.78\textwidth]{../final_report/figures/q2_symmetric_poisson.png}

\vspace{0.25ex}
\scriptsize\textbf{Avec couplage :} $\lambda_2^+$ excitée par le niveau 1, moyennes $\approx 10$--$11$\\[-0.2ex]
\includegraphics[width=0.78\textwidth]{../final_report/figures/q2_symmetric_excited.png}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Passage au prix : dérive induite par l'asymétrie}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.40\textwidth}
\small
Modèle à 4 limites : à chaque épuisement, le prix saute de $\pm\delta^p/2$, et les intensités Hawkes sont \alert{transportées} d'un cycle à l'autre.
\[
\Delta p_0=\tfrac{\delta^p}{2}(2p-1),\quad p=\PP(\taua<\taub).
\]

\vspace{0.5ex}
\begin{block}{Lecture}
\small
\begin{itemize}
\item $p=\tfrac12$ : prix martingale.
\item $p>\tfrac12$ : dérive haussière.
\item $p<\tfrac12$ : dérive baissière.
\end{itemize}
\end{block}

\vspace{0.2ex}
Le cas symétrique sert de référence ; l'asymétrie $\mu_a^-\neq\mu_b^-$ crée une tendance directionnelle.
\end{column}
\begin{column}{0.58\textwidth}
\centering
\includegraphics[width=\textwidth]{mp_sym.png}\\
{\scriptsize Cas symétrique : pas de tendance (martingale).}

\vspace{0.5ex}
\includegraphics[width=\textwidth]{mp_asym.png}\\
{\scriptsize Cas asymétrique : tendance haussière nette.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Réduire le coût de simulation}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Avant les cascades rares.} On étudie d'abord un cas de contrôle, avec probabilité \alert{$p\simeq0{,}14$}, où le Monte-Carlo direct reste disponible comme référence.\\[1ex]

\textbf{Objectif.} Comparer MC direct et AMS à précision comparable. Le MC direct relance chaque trajectoire depuis zéro ; le splitting AMS \alert{recycle} celles qui ont déjà atteint des paliers intermédiaires.

\vspace{1ex}
Résultat attendu : \alert{même précision, moins de simulations}.
\end{column}
\begin{column}{0.45\textwidth}
\begin{block}{Benchmark contrôlé}
\begin{itemize}
\item MC direct : $N=10^5$.
\item Splitting AMS : $N=3\cdot10^4$.
\item Même loi de $Q_2(\tau)$ : forme et support conservés.
\end{itemize}
\end{block}
\vspace{1ex}
\centering\footnotesize
Une fois validé sur $p\simeq0{,}14$, le même outil est utilisé pour les cascades vraiment rares.
\end{column}
\end{columns}
\end{frame}

%----------------------------------------------------------------
\begin{frame}{Splitting AMS : recycler les trajectoires}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Principe.} Plutôt que de relancer $N$ trajectoires depuis zéro, on conserve celles qui ont déjà atteint le palier précédent : elles sont rééchantillonnées et seul le tronçon restant est simulé.\\[1ex]

Sur le benchmark modéré, cela mesure le gain de budget. Pour les flash crashes, cela revient ensuite à décomposer une probabilité infime en produit de probabilités conditionnelles, en transportant l'\alert{état Hawkes complet}.
\end{column}
\begin{column}{0.52\textwidth}
\centering
\includegraphics[width=\textwidth]{fig_ams_conv.png}

\vspace{0.3ex}
{\footnotesize Écart-type de $\hat P_{\text{AMS}}$ sur le benchmark $p\simeq0{,}14$ : la droite log-log valide la CLT.}

\vspace{0.5ex}
\small\textbf{Validation :} pente log-log $\approx{-0{,}5}$~; CV réduit d'un facteur $7$ entre $N=50$ et $N=3200$.
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Loi de $Q_2$ : contrôle par Monte-Carlo direct}
\begin{columns}[T]
\begin{column}{0.40\textwidth}
\small
La loi de $Q_{a,2}(\tau)$ conditionne l'état initial du cycle suivant. Pour vérifier que le splitting ne déforme pas cette loi, on compare deux estimateurs.\\[0.6ex]

\begin{itemize}
\item \textbf{MC direct} ($N{=}10^5$) : référence brute — la plupart des trajectoires ne conditionnent pas sur $\tau_a<\tau_b$.
\item \textbf{Splitting AMS} ($N{=}3\!\cdot\!10^4$) : seules les trajectoires recyclées contribuent.
\end{itemize}
\textbf{Résultat :} même distribution avec $\mathbf{3\times}$ moins de calcul. Validation empirique que le recyclage ne déforme pas la loi conditionnelle.
\end{column}
\begin{column}{0.60\textwidth}
\centering
\includegraphics[width=\textwidth]{q2_mc_ams.png}

\vspace{0.5ex}
{\footnotesize Loi de $Q_{a,2}(\tau_a)$ : Monte-Carlo direct ($N{=}10^5$, g.) vs splitting AMS ($N{=}3\!\cdot\!10^4$, d.). Même forme, même support.}
\end{column}
\end{columns}
\end{frame}

%----------------------------------------------------------------
\begin{frame}{Flash crash de profondeur $k$}
\begin{columns}[T]
\begin{column}{0.44\textwidth}
\small
On appelle flash crash de profondeur $k$ une série de $k$ épuisements \emph{bid} avant qu'un épuisement ask ne casse la séquence :
\[
\mathrm{FC}_k=\{\taub^{(1)}<\taua,\ldots,\taub^{(k)}<\taua\}.
\]

Les cycles ne sont pas i.i.d. : après chaque saut, les volumes reconstruits et les intensités Hawkes sont transportés.
\[
\begin{aligned}
\PP(\mathrm{FC}_k)&=\prod_{i=1}^{k} c_i,\\
c_i&=\PP(\taub^{(i)}<\taua\mid \mathrm{FC}_{i-1}).
\end{aligned}
\]

\textbf{Lecture :} au-delà de $k=3$, l'événement devient invisible au MC direct ; le splitting AMS permet de poursuivre l'estimation.
\end{column}
\begin{column}{0.56\textwidth}
\centering
\includegraphics[width=0.96\textwidth]{../final_report/figures/flash_crash_asym.png}

\vspace{0.3ex}
{\scriptsize
\textbf{Ordres de grandeur.} Régime bid rare : $\mu_a^-=3{,}0$, $\mu_b^-=1{,}2$, $q_{a,0}=10$, $q_{b,0}=12$.\\[0.3ex]
$\PP(\mathrm{FC}_1)\simeq1{,}4\cdot10^{-1}$ ; $\PP(\mathrm{FC}_5)\simeq1{,}1\cdot10^{-7}$ ; $\PP(\mathrm{FC}_8)\simeq1{,}9\cdot10^{-13}$.
}
\end{column}
\end{columns}
\end{frame}

%----------------------------------------------------------------
\begin{frame}{Splitting en deux phases}
\begin{columns}[T]
\begin{column}{0.40\textwidth}
\small
\textbf{Idée.} Les premiers paliers de l'AMS ne dépendent pas de la cible finale et sont \alert{réutilisables}. On les calcule une fois hors-ligne, on stocke $\hat\pi_{k^*}$, puis on ne simule que les derniers paliers en ligne :
\[
\resizebox{\linewidth}{!}{$\displaystyle
\hat P(\tau_s<\tau_{\bar s})=
\underbrace{\prod_{j=Q_0-1}^{k^*+1}\hat p_j}_{P_{\mathrm{off}}}
\times
\hat P\!\bigl(\tau_s<\tau_{\bar s}\mid q_{s,1}\le k^*\bigr)$}
\]
\textbf{Gain de calcul :} splitting complet $= 7{,}69$\,s par requête ; avec deux phases : $7{,}69$\,s une fois hors-ligne, puis $0{,}67$\,s par requête en ligne ($\times\mathbf{11}$ plus rapide). Moyennes $12{,}90$ vs $12{,}91$.
\end{column}
\begin{column}{0.60\textwidth}
\centering
\includegraphics[width=\textwidth]{q2_boot.png}

\vspace{0.5ex}
{\scriptsize Loi de $Q_{a,2}$ : splitting complet (g.) vs bootstrap depuis $\hat\pi_{k^*=5}$ (d.).\\
Moyennes $12{,}90$ vs $12{,}91$.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Importance sampling : mesure et poids}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\vspace{3ex}
\textbf{Principe.} Au lieu de sélectionner les trajectoires par niveaux, on \alert{modifie la loi simulée} pour rendre l'événement plus fréquent, puis on corrige par un poids de vraisemblance. Le paramètre $\theta$ règle l'intensité de la déformation.\\[1ex]

\begin{itemize}
\item $\theta^*\approx 0{,}30$ : meilleur réglage observé.
\item $\theta>0{,}6$ : les poids deviennent instables, avec un CV qui dépasse $100$.
\end{itemize}

\end{column}

\centering
\includegraphics[width=\textwidth]{fig_is.png}

\vspace{0.5ex}

\end{columns}
\end{frame}

\end{frame}

\begin{frame}{Conclusion} \begin{columns}[T] \begin{column}{0.6\textwidth} \begin{enumerate} \item[\textbf{I.}] La mémoire Hawkes réduit les temps d'épuisement et modifie la loi complète via le mécanisme d'auto-excitation. \item[\textbf{II.}] AMS sert d'abord à économiser le budget sur un benchmark modéré, puis à traiter les cascades rares ; IS sert de comparaison indépendante quand les poids restent stables. \item[\textbf{III.}] Sur Black Thursday, le régime pré-crise calibré est quasi-Poisson ; la séquence observée reste rare dans ce régime, ce qui illustre le risque de sous-estimer une crise avec une calibration calme. \end{enumerate} \end{column} \begin{column}{0.4\textwidth} \begin{block}{Perspectives} \begin{itemize} \item Données tick-by-tick (TARDIS/Kaiko) pour mieux séparer $\alpha,\beta$. \item Extension à $L>2$ niveaux. \item Résilience du carnet : réapparition possible des limites consommées. \end{itemize} \end{block} \end{column} \end{columns} \end{frame}

\begin{frame}{Annexe A : Thinning d'Ogata}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\small
\begin{algorithmic}[1]
\State $t\gets0$,\; $q\gets q_0$,\; $\lambda^-\gets\mu^-$
\While{condition d'arrêt non atteinte}
  \State $\bar\lambda\gets\max(\mu^-,\lambda^-)$,\;
         $\Lambda\gets\bar\lambda+\lambda^+$
  \State $\Delta t\sim\mathcal{E}(\Lambda)$,\; $t'\gets t+\Delta t$
  \State $\lambda^-_\mathrm{pre}\gets\mu^-+(\lambda^--\mu^-)e^{-\beta(t'-t)}$
  \State $U\sim\mathcal{U}[0,1]$
  \If{$U<\lambda^+/\Lambda$}
    \State $q\gets q+1$,\;
           $\lambda^-\gets\max(\lambda^-_\mathrm{pre}-\alpha,0)$
           \Comment{insertion : inhibition}
  \ElsIf{$U<(\max(\lambda^-_\mathrm{pre},0)+\lambda^+)/\Lambda$}
    \State $q\gets q-1$,\;
           $\lambda^-\gets\lambda^-_\mathrm{pre}+\alpha$
           \Comment{annulation : auto-excitation}
  \Else\; $\lambda^-\gets\lambda^-_\mathrm{pre}$
    \Comment{rejet}
  \EndIf
  \State $t\gets t'$
\EndWhile
\end{algorithmic}
\end{column}
\begin{column}{0.45\textwidth}
\small
\textbf{Principe.} $\Lambda$ est une borne sur l'intensité totale $\lambda^-(t)+\lambda^+$. On propose le prochain événement à $t+\Delta t$ avec $\Delta t\sim\mathcal{E}(\Lambda)$, et on l'accepte ou le rejette par thinning.\\[1ex]

\textbf{Point de vigilance.} L'inhibition par insertion ($\lambda^-\gets\max(\cdot-\alpha,0)$) impose un plancher à 0. Le bilan de flux théorique n'est plus exactement conservé~; l'intensité stationnaire mesurée ($2{,}96$) dépasse légèrement la valeur théorique ($2{,}80$).
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Annexe B : Algorithme AMS}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\small
\textbf{Entrée :} $N$ particules, $Q_0$ niveaux, côté cible $s$.
\begin{algorithmic}[1]
\State Initialiser $N$ particules à l'état $(q_{s,0},\lambda_a^-,\lambda_b^-)$ initial
\For{$k = Q_0-1$ \textbf{downto} $0$}
  \State Simuler chaque particule par thinning jusqu'à $q_{s,1}=k$
         (succès) ou $q_{\bar s,1}=0$ (échec)
  \State $n_k\gets$ nb de succès ;\; $\hat p_k\gets n_k/N$
  \State Rééchantillonner $N$ copies parmi les $n_k$ survivants
         (avec remise), en héritant de $(\lambda_a^-,\lambda_b^-,t)$ finals
\EndFor
\State \Return $\hat P_\mathrm{AMS}=\prod_{k=0}^{Q_0-1}\hat p_k$
\end{algorithmic}
\end{column}
\begin{column}{0.48\textwidth}
\small
\textbf{CLT (méthode delta).} Pour $N\to\infty$ :
\[
\sqrt{N}(\hat P_\mathrm{AMS}-P)\xrightarrow{d}
\mathcal{N}\!\Bigl(0,P^2\!\sum_k\tfrac{1-p_k}{p_k}\Bigr).
\]
\textbf{Preuve.} $\hat p_k=n_k/N$ avec $n_k\sim\mathrm{Bin}(N,p_k)$. Par indépendance asymptotique des niveaux et méthode delta sur $\log\hat P=\sum_k\log\hat p_k$ :
\[
\Var(\hat P)\approx P^2\sum_k\frac{1-p_k}{N p_k}.
\]
L'IC à 95\,\% est $\hat P_\mathrm{AMS}\pm1{,}96\,\hat P_\mathrm{AMS}\sqrt{\frac{1}{N}\sum_k\frac{1-\hat p_k}{\hat p_k}}$.
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Annexe C : AMS deux phases}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\small
\textbf{Phase hors-ligne} (calculée une fois).
\begin{algorithmic}[1]
\State Lancer AMS de $k=Q_0-1$ jusqu'à $k=k^*+1$
\State Stocker les états survivants :
       $\hat\pi_{k^*}=\{s^{(i)}_{k^*}\}_{i=1}^{N}$
\State $P_\mathrm{off}\gets\prod_{k=k^*+1}^{Q_0-1}\hat p_k$
\end{algorithmic}

\vspace{1ex}
\textbf{Phase en ligne} (par requête, $7{,}69\to 0{,}67$\,s).
\begin{algorithmic}[1]
\State Tirer $N$ états i.i.d.\ depuis $\hat\pi_{k^*}$
\State Lancer AMS de $k=k^*$ jusqu'à $0$
\State $P_\mathrm{on}\gets\prod_{k=0}^{k^*}\hat p_k$
\State \Return $P_\mathrm{off}\times P_\mathrm{on}$
\end{algorithmic}
\end{column}
\begin{column}{0.50\textwidth}
\small
\textbf{Formule.}
\[
\hat P(\tau_s<\tau_{\bar s})=
\underbrace{\prod_{k=k^*+1}^{Q_0-1}\hat p_k}_{P_\mathrm{off}}
\times
\hat P\!\bigl(\tau_s<\tau_{\bar s}\mid q_{s,1}\le k^*\bigr).
\]

\textbf{Validité.} L'estimateur reste non-biaisé si $\hat\pi_{k^*}$ est calculé sur une phase indépendante. La convergence reste en $1/\sqrt N$ pour la phase en ligne.

\vspace{0.5ex}
\textbf{Gain.} ${\times}11$ par requête après un coût hors-ligne de $7{,}69$\,s ($k^*{=}5$, $N{=}50\,000$).
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Annexe D : Importance sampling --- démonstration}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\small
\textbf{Changement de mesure.} On tord les intensités par $\theta>0$ :
\[
\tilde\lambda_a=e^{-\theta}\lambda_a,\quad
\tilde\lambda_b=e^{+\theta}\lambda_b.
\]
L'estimateur IS est $\hat p_\mathrm{IS}=N^{-1}\sum_i L_i\,\mathbf{1}_{\tau_b<\tau_a}$ sous $\tilde P_\theta$.

\begin{block}{Proposition (log-rapport de vraisemblance)}
\small
\[
\log L = (e^{-\theta}-1)I_a+(e^{\theta}-1)I_b+\theta N_a-\theta N_b,
\]
où $I_a=\int_0^\tau\!\lambda_a\,dt$ est le compensateur ask et $N_a,N_b$ les comptages.
\end{block}
\end{column}
\begin{column}{0.50\textwidth}
\small
\textbf{Preuve (Girsanov pour processus ponctuels).}
\[
\log\frac{dP}{d\tilde P}=
\int_0^\tau\!\log\frac{\lambda_a}{\tilde\lambda_a}\,dN_a
+\int_0^\tau\!(\tilde\lambda_a-\lambda_a)\,dt+\text{bid}.
\]
En substituant $\tilde\lambda_a=e^{-\theta}\lambda_a$ :
\begin{itemize}\small
  \item terme événementiel~: $\log(e^\theta)\,N_a = +\theta N_a$
  \item compensateur~: $(e^{-\theta}-1)I_a$
\end{itemize}
Le terme bid donne symétriquement $-\theta N_b+(e^\theta-1)I_b$.\\[0.5ex]
\textbf{Optimal.} $\theta^*\approx0{,}30$ minimise $\mathrm{CV}^2(\theta)$. Au-delà de $\theta=0{,}6$, les poids dégénèrent (CV$\to100$).
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Annexe E : Calibration MLE --- démonstration}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.55\textwidth}
\footnotesize
Modèle réduit (excitation croisée) :
\[
\lambda_a(t)=\mu_a+\alpha\bigl(R_a(t)+R_b(t)\bigr),
\]
avec la récurrence $R_a(t_i)=e^{-\beta(t_i-t_{i-1})}R_a(t_{i-1})+\mathbf{1}_{t_i\in\mathcal{D}_a}$.

\begin{block}{Proposition (log-vraisemblance $O(n)$)}
\scriptsize
\begin{align*}
\ell &= \sum_{t\in\mathcal{D}_a}\!\log(\mu_a+\alpha(R_a+R_b))\\
     &+\sum_{t\in\mathcal{D}_b}\!\log(\mu_b+\alpha(R_b+R_a))\\
     &-(\mu_a+\mu_b)T
      -\frac{2\alpha}{\beta}(S_a+S_b),
\end{align*}
$S_s=\sum_{u\in\mathcal{D}_s}(1-e^{-\beta(T-u)})$.
\end{block}
\end{column}
\begin{column}{0.39\textwidth}
\scriptsize
\vspace{3ex}
\textbf{Preuve (compensateur).}\\[-0.2ex]
Pour un processus ponctuel :
\[
\ell=\sum_i\log\lambda(t_i)-\int_0^T\lambda(t)\,dt.
\]
Le compensateur total vaut :
\[
\int_0^T\!\lambda_a\,dt=\mu_a T+\frac{\alpha}{\beta}(S_a+S_b).
\]
En additionnant ask et bid, les termes croisés s'additionnent, d'où $-2\alpha(S_a+S_b)/\beta$.

\vspace{0.5ex}

MLE  & 3{,}01 & 1{,}50 & 0{,}37 & 1{,}15\\
Err. & $<1\%$ & $<1\%$ & $6\%$ & $15\%$\\
\bottomrule
\end{tabular}
\end{center}
\end{column}
